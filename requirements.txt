# 5-1 2FA: QR-code rendering only (TOTP/HOTP itself is hand-rolled against
# RFC 4226/6238 test vectors in totp.py using stdlib hmac/struct/base64 --
# no reason to add a dependency for ~30 lines of well-specified math). QR
# encoding is a different story (Reed-Solomon + mask-pattern scoring is easy
# to get subtly wrong in a way that still *looks* like a QR code but won't
# scan) -- qrcode is pure Python, its SVG output path needs no Pillow, and
# its only dependency is colorama (already transitively present).
qrcode==8.0
# 3-2 보고서 내보내기: PDF/Excel generation. Both pure-Python-friendly on
# Windows (reportlab ships prebuilt wheels for its Pillow dependency, no
# build toolchain needed) -- hand-rolling either file format is far riskier
# than the QR-encoding precedent above (a PDF with broken table/page-break
# logic, or an XLSX with a malformed zip/XML structure, fails in ways that
# are hard to catch without opening the file in real Acrobat/Excel).
reportlab==5.0.1
openpyxl==3.1.5
# Windows 이벤트 로그 에러 조회: psutil has no Event Log support at all (it's a
# Windows-specific API, same reason win_service_iter() for the services card
# needed no extra dependency -- psutil calls the Service Control Manager
# directly via ctypes, but the Event Log API has no such built-in path).
# pywin32's win32evtlog wraps the legacy (pre-Vista, still fully supported)
# OpenEventLog/ReadEventLog API -- simpler and more robust for "give me the
# last N error/warning records" than the newer XML-query EvtQuery API would
# be. Windows-only: local_collector.py/agent.py both already guard every
# Windows-only sampler behind IS_WINDOWS (see sample_services()), same
# pattern here. Existing deployments (this PC, the remote center PC) need
# `pip install -r requirements.txt` re-run to pick this up; agent-mode
# machines get it for free in the next InfraSightAgent.exe rebuild (PyInstaller
# bundles it into the exe, nothing to install on the target PC itself).
pywin32==311

# 장비의 CLI(enable 모드, interface 서브모드, reload/reboot 등)로 가야 한다.
# netmiko를 고른 이유: paramiko(저수준 SSH 전송)만으로는 벤더별 enable 모드
# 진입, 프롬프트 인식, "Proceed with reload? [confirm]" 같은 대화형 확인까지
# 전부 직접 손으로 처리해야 하는데, netmiko는 device_type별로(cisco_ios/

# 처리해주고, 이 프로젝트의 snmp_profiles.py 벤더 프로필 구조와 1:1로
# 맞물린다. SSH/Telnet 둘 다 지원(디바이스마다 '_telnet' 접미사 device_type).
netmiko==4.8.0
